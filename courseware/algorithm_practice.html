一、字符串 / 数组

1.1 反转字符串

双指针从两端向中间交换字符。

void reverse(char *s, int len) {
    int i = 0, j = len - 1;
    while (i < j) {
        char t = s[i]; s[i] = s[j]; s[j] = t;
        i++; j--;
    }
}

1.2 实现 atoi

跳过前导空格，处理正负号，逐位转数字，注意溢出。

int my_atoi(const char *s) {
    int sign = 1, res = 0;
    while (*s == ' ') s++;
    if (*s == '-' || *s == '+') {
        sign = (*s == '-') ? -1 : 1;
        s++;
    }
    while (*s >= '0' && *s <= '9') {
        int d = *s - '0';
        // 简单溢出检查
        if (res > (INT_MAX - d) / 10) return sign == 1 ? INT_MAX : INT_MIN;
        res = res * 10 + d;
        s++;
    }
    return sign * res;
}

1.3 二分查找

int binary_search(int *arr, int n, int target) {
    int l = 0, r = n - 1;
    while (l <= r) {
        int m = l + (r - l) / 2;
        if (arr[m] == target) return m;
        if (arr[m] < target) l = m + 1;
        else r = m - 1;
    }
    return -1;
}

1.4 两数之和（返回索引，数组已排序）

双指针，和小了左移，大了右移。

void two_sum(int *arr, int n, int target, int *out) {
    int l = 0, r = n - 1;
    while (l < r) {
        int sum = arr[l] + arr[r];
        if (sum == target) { out[0] = l; out[1] = r; return; }
        if (sum < target) l++;
        else r--;
    }
    out[0] = out[1] = -1;
}

1.5 寻找缺失数字（0..n 中缺一个）

利用下标和值的对应关系，用异或或求和。

int missing_number(int *arr, int n) {
    int x = 0;
    for (int i = 0; i < n; i++) x ^= (i + 1) ^ arr[i];
    return x;
}

二、链表

2.1 反转单链表

struct ListNode { int val; struct ListNode *next; };

struct ListNode* reverse_list(struct ListNode *head) {
    struct ListNode *prev = NULL, *cur = head;
    while (cur) {
        struct ListNode *next = cur->next;
        cur->next = prev;
        prev = cur;
        cur = next;
    }
    return prev;
}

2.2 判断链表是否有环

快慢指针，相遇则有环。

int has_cycle(struct ListNode *head) {
    struct ListNode *slow = head, *fast = head;
    while (fast && fast->next) {
        slow = slow->next;
        fast = fast->next->next;
        if (slow == fast) return 1;
    }
    return 0;
}

2.3 合并两个有序链表

struct ListNode* merge_two_lists(struct ListNode *l1, struct ListNode *l2) {
    struct ListNode dummy, *p = &dummy;
    dummy.next = NULL;
    while (l1 && l2) {
        if (l1->val < l2->val) { p->next = l1; l1 = l1->next; }
        else { p->next = l2; l2 = l2->next; }
        p = p->next;
    }
    p->next = l1 ? l1 : l2;
    return dummy.next;
}

三、栈与队列

3.1 用两个栈实现队列

#define MAX 100
int s1[MAX], s2[MAX];
int top1 = -1, top2 = -1;

void push(int x) { s1[++top1] = x; }

int pop() {
    if (top2 == -1) {
        while (top1 >= 0) s2[++top2] = s1[top1--];
    }
    if (top2 == -1) return -1;  // 空
    return s2[top2--];
}

3.2 最小栈

用一个辅助栈保存当前最小值。

int data[MAX], minstk[MAX];
int top = -1, mtop = -1;

void min_push(int x) {
    data[++top] = x;
    if (mtop == -1 || x <= minstk[mtop]) minstk[++mtop] = x;
}

int min_pop() {
    if (top == -1) return -1;
    int v = data[top--];
    if (v == minstk[mtop]) mtop--;
    return v;
}

int get_min() { return mtop == -1 ? -1 : minstk[mtop]; }

四、位运算

4.1 判断是否为 2 的幂

int is_power_of_two(int n) {
    return n > 0 && (n & (n - 1)) == 0;
}

4.2 统计二进制 1 的个数

int count_ones(unsigned int n) {
    int c = 0;
    while (n) {
        n &= n - 1;  // 清除最低位的 1
        c++;
    }
    return c;
}

4.3 不用临时变量交换两个数

void swap(int *a, int *b) {
    if (a == b) return;
    *a = *a ^ *b;
    *b = *a ^ *b;
    *a = *a ^ *b;
}
注意：异或交换在嵌入式中可读性差，面试写出思路即可，实际代码建议用临时变量。

五、排序

5.1 快速排序

void quick_sort(int *arr, int l, int r) {
    if (l >= r) return;
    int pivot = arr[l], i = l, j = r;
    while (i < j) {
        while (i < j && arr[j] >= pivot) j--;
        while (i < j && arr[i] <= pivot) i++;
        if (i < j) { int t = arr[i]; arr[i] = arr[j]; arr[j] = t; }
    }
    arr[l] = arr[i]; arr[i] = pivot;
    quick_sort(arr, l, i - 1);
    quick_sort(arr, i + 1, r);
}

5.2 归并排序

void merge(int *arr, int l, int m, int r, int *tmp) {
    int i = l, j = m + 1, k = 0;
    while (i <= m && j <= r) tmp[k++] = (arr[i] <= arr[j]) ? arr[i++] : arr[j++];
    while (i <= m) tmp[k++] = arr[i++];
    while (j <= r) tmp[k++] = arr[j++];
    for (i = 0; i < k; i++) arr[l + i] = tmp[i];
}

void merge_sort(int *arr, int l, int r, int *tmp) {
    if (l >= r) return;
    int m = l + (r - l) / 2;
    merge_sort(arr, l, m, tmp);
    merge_sort(arr, m + 1, r, tmp);
    merge(arr, l, m, r, tmp);
}

6.2 约瑟夫环

递推公式：f(n, m) = (f(n-1, m) + m) % n。

int josephus(int n, int m) {
    int res = 0;  // f(1, m) = 0
    for (int i = 2; i <= n; i++)
        res = (res + m) % i;
    return res;
}

6.3 斐波那契数列

int fib(int n) {
    if (n <= 1) return n;
    int a = 0, b = 1;
    for (int i = 2; i <= n; i++) {
        int t = a + b;
        a = b;
        b = t;
    }
    return b;
}

面试手撕要点

  • 先讲思路：时间复杂度、空间复杂度、边界条件。
  • 边写边说：面试官听的是你的思考过程，不是只看书写结果。
  • 边界检查：空指针、空数组、长度为 1、最大/最小值、溢出。
  • 模块化：先写框架，再填细节；能用辅助函数/哨兵节点就别硬写。
  • 嵌入式常考：指针操作、位运算、有限状态机、数组/字符串、链表。
