15. 3Sum

排序后固定一个数，双指针找和为 0 的两数，注意去重。

int cmp(const void *a, const void *b) {
    return *(int *)a - *(int *)b;
}

void threeSum(int *nums, int n) {
    qsort(nums, n, sizeof(int), cmp);
    for (int i = 0; i < n - 2; i++) {
        if (i > 0 && nums[i] == nums[i - 1]) continue;
        int l = i + 1, r = n - 1;
        while (l < r) {
            int sum = nums[i] + nums[l] + nums[r];
            if (sum == 0) {
                printf("%d %d %d\n", nums[i], nums[l], nums[r]);
                while (l < r && nums[l] == nums[l + 1]) l++;
                while (l < r && nums[r] == nums[r - 1]) r--;
                l++; r--;
            } else if (sum < 0) {
                l++;
            } else {
                r--;
            }
        }
    }
}

42. Trapping Rain Water

双指针从两端向中间移动，分别维护左右最高高度。

int trap(int *height, int n) {
    int l = 0, r = n - 1;
    int lmax = 0, rmax = 0, res = 0;
    while (l < r) {
        if (height[l] < height[r]) {
            if (height[l] >= lmax) lmax = height[l];
            else res += lmax - height[l];
            l++;
        } else {
            if (height[r] >= rmax) rmax = height[r];
            else res += rmax - height[r];
            r--;
        }
    }
    return res;
}

33. Search in Rotated Sorted Array

二分查找，判断 mid 所在的有序半边。

int search(int *nums, int n, int target) {
    int l = 0, r = n - 1;
    while (l <= r) {
        int m = l + (r - l) / 2;
        if (nums[m] == target) return m;
        if (nums[l] <= nums[m]) {
            if (target >= nums[l] && target < nums[m])
                r = m - 1;
            else
                l = m + 1;
        } else {
            if (target > nums[m] && target <= nums[r])
                l = m + 1;
            else
                r = m - 1;
        }
    }
    return -1;
}

200. Number of Islands

DFS 遇到 1 就淹没整个连通区域。

void dfs(char **grid, int m, int n, int i, int j) {
    if (i < 0 || i >= m || j < 0 || j >= n || grid[i][j] != '1')
        return;
    grid[i][j] = '0';
    dfs(grid, m, n, i + 1, j);
    dfs(grid, m, n, i - 1, j);
    dfs(grid, m, n, i, j + 1);
    dfs(grid, m, n, i, j - 1);
}

int numIslands(char **grid, int m, int n) {
    int cnt = 0;
    for (int i = 0; i < m; i++)
        for (int j = 0; j < n; j++)
            if (grid[i][j] == '1') {
                cnt++;
                dfs(grid, m, n, i, j);
            }
    return cnt;
}

76. Minimum Window Substring

滑动窗口，need 计数和 have 计数，右扩左缩。

char *minWindow(char *s, char *t) {
    int need[128] = {0}, have[128] = {0};
    int required = 0, formed = 0;
    for (char *p = t; *p; p++) {
        if (need[(unsigned char)*p] == 0) required++;
        need[(unsigned char)*p]++;
    }

    int start = 0, len = 0x7fffffff;
    int l = 0;
    for (int r = 0; s[r]; r++) {
        char c = s[r];
        have[(unsigned char)c]++;
        if (have[(unsigned char)c] == need[(unsigned char)c]) formed++;

        while (formed == required) {
            if (r - l + 1 < len) {
                start = l;
                len = r - l + 1;
            }
            char left = s[l];
            have[(unsigned char)left]--;
            if (have[(unsigned char)left] < need[(unsigned char)left])
                formed--;
            l++;
        }
    }
    if (len == 0x7fffffff) {
        char *empty = malloc(1);
        empty[0] = '\0';
        return empty;
    }
    char *res = malloc(len + 1);
    memcpy(res, s + start, len);
    res[len] = '\0';
    return res;
}

1143. Longest Common Subsequence

二维 DP，相等时左上 +1，否则取左/上最大值。

int longestCommonSubsequence(char *a, char *b) {
    int m = strlen(a), n = strlen(b);
    int dp[m + 1][n + 1];
    memset(dp, 0, sizeof(dp));
    for (int i = 1; i <= m; i++) {
        for (int j = 1; j <= n; j++) {
            if (a[i - 1] == b[j - 1])
                dp[i][j] = dp[i - 1][j - 1] + 1;
            else
                dp[i][j] = dp[i - 1][j] > dp[i][j - 1] ? dp[i - 1][j] : dp[i][j - 1];
        }
    }
    return dp[m][n];
}

322. Coin Change

完全背包，dp[i] 表示凑成金额 i 所需最少硬币数。

int coinChange(int *coins, int n, int amount) {
    int dp[amount + 1];
    for (int i = 0; i <= amount; i++) dp[i] = amount + 1;
    dp[0] = 0;
    for (int i = 1; i <= amount; i++) {
        for (int j = 0; j < n; j++) {
            if (coins[j] <= i) {
                int cand = dp[i - coins[j]] + 1;
                if (cand < dp[i]) dp[i] = cand;
            }
        }
    }
    return dp[amount] > amount ? -1 : dp[amount];
}
